## Contracts

**אירוע מסקריפט לסקריפט ה-CRM** — `POST {action:'event', secret, event}`:
- ליד: `{type:'lead', name, business, phone, email, message, subject, page, landing, source, gclid}`
- הזמנה / בקשה: `{type:'order'|'quote_request', via:'אתר'|'דף הסוכן', orderId, business, contact, phone, address, totalItems, totalPrice, pdfUrl, quoteLink}`
- תשובה: `{ok:true, id}` · `{ok:true, id, duplicate:true}` (מספר הזמנה שכבר רשום) · `{ok:false, error}`

**מהעמוד** — `POST {action, password, ...}`, גוף JSON ב-`text/plain`:
| action | בקשה | תשובה |
|---|---|---|
| `list` | — | `{ok, today, contacts:[Contact], activities:[Activity]}` |
| `add` | `{contact:{phone, name, business, source, followUp, note}}` | `{ok, contact, activity}` או `{ok:false, error:'duplicate', id}` |
| `update` | `{id, fields}` | `{ok, contact, activity:null}` |
| `log` | `{id, fields, activity:{type, text}}` | `{ok, contact, activity}` |
| `delete` | `{id}` | `{ok, deleted}` |
| `import` | `{dryRun}` (ברירת מחדל true) | `{ok, dryRun, summary:{rows, created, updated, activities, skipped:{chain, committee, consumer, test, noKey, unknown, existing}, testRows:[...]}}` |

שגיאות: `unauthorized`, `bad request`, `bad phone`, `name required`, `bad date`, `bad stage`, `bad activity`, `duplicate`, `not found`, `busy`, `server error`.

**Contact:** `{id, name, business, phone, email, address, stage, followUp, source, created, lastActivity, orders, lastOrder}` — הכל מחרוזות. `phone` ספרות בלבד, `followUp`/`lastOrder` ‏`YYYY-MM-DD`, `created`/`lastActivity` ‏`YYYY-MM-DD HH:mm`.
**Activity:** `{id, contactId, at, type, text, orderId, pdf, quoteLink, by}`.
**שלבים:** `חדש` · `חזרתי` · `נשלחה הצעה` · `לקוח` · `לא רלוונטי`.

function applyEvent_(model, ev, now) {

  if (orderId) {
    for (var i = 0; i < model.activities.length; i++) {
      if (model.activities[i].orderId === orderId) {
        return { skipped: true, contact: contactById_(model, model.activities[i].contactId) };
      }
    }

  var phone = normalizePhone_(ev.phone);
  var business = trim_(ev.business);
  var name = trim_(ev.name || ev.contact);
  var c = findContact_(model.contacts, phone, business || name);
  var created = false;
  if (!c) {
    c = blankContact_(at);
    c.source = sourceFor_(ev);
    model.contacts.push(c);
    created = true;
  }
  // משלימים רק שדות ריקים — מה שרועי תיקן בכרטיס לא נדרס. הכתובת היא חריג:
  // היא מתעדכנת להזמנה האחרונה.
  if (!c.phone && phone) c.phone = phone;
  if (!c.name && name) c.name = name;
  if (!c.business && business) c.business = business;
  if (!c.email && trim_(ev.email)) c.email = trim_(ev.email);
  if (trim_(ev.address)) c.address = trim_(ev.address);

  if (ev.type === 'order') {
    c.stage = 'לקוח';
    c.orders = String((Number(c.orders) || 0) + 1);
    if (!c.lastOrder || at.slice(0, 10) > c.lastOrder) c.lastOrder = at.slice(0, 10);
    c.followUp = '';
  } else {
    // שלב לא יורד לבד; רק מי שסומן "לא רלוונטי" נפתח מחדש
    if (c.stage === 'לא רלוונטי') c.stage = 'חדש';
    // פנייה חדשה מחזירה ל"לטיפול" — אבל תאריך שכבר עבר נשאר (הוא מוקדם יותר)
    if (!ev.imported && (!c.followUp || c.followUp > now.date)) c.followUp = now.date;
  }
  if (!c.lastActivity || at > c.lastActivity) c.lastActivity = at;
  if (!c.created || at < c.created) c.created = at;
  var activity = activityFor_(ev, c.id, at);
  model.activities.push(activity);
  return { contact: c, activity: activity, created: created };
}

// שורה מגיליון ההזמנות: { kind: 'quote_request' }, { kind: 'order', via }
// או { kind: 'skip', reason }. זה CRM לעסקים — רשת, ועד וצרכן לא נכנסים.
function classifyOrderRow_(r) {
  var prefix = trim_(r.orderId).slice(0, 3);
  if (prefix === 'RS-' || prefix === 'RA-') return { kind: 'skip', reason: 'chain' };
  if (prefix === 'VD-') return { kind: 'skip', reason: 'committee' };
  if (/בדיקה|למחוק/.test([trim_(r.business), trim_(r.contact), trim_(r.notes)].join(' '))) {
    return { kind: 'skip', reason: 'test' };
  }
  // הזמנת צרכן מ-/care/: יש לה שיטת אספקה ואין לה שם עסק
  if (prefix === 'AR-' && (trim_(r.shipping) || !trim_(r.business))) {
    return { kind: 'skip', reason: 'consumer' };
  }
  if (['HM-', 'AR-', 'SL-'].indexOf(prefix) < 0) return { kind: 'skip', reason: 'unknown' };
  if (!normalizePhone_(r.phone) && !trim_(r.business)) return { kind: 'skip', reason: 'noKey' };
  if (prefix === 'HM-') return { kind: 'quote_request' };
  return { kind: 'order', via: prefix === 'SL-' ? 'דף הסוכן' : 'אתר' };
}

// ייבוא מגיליון ההזמנות, מהישן לחדש. מחזיר { summary, touched } — touched הם
// מזהי אנשי הקשר שנוצרו או השתנו, כדי שהשמירה תכתוב רק אותם.
function buildImport_(model, rows, now) {
  var summary = {
    rows: rows.length, created: 0, updated: 0, activities: 0,
    skipped: { chain: 0, committee: 0, consumer: 0, test: 0, noKey: 0, unknown: 0, existing: 0 },
    testRows: []
  };
  var touched = {};
  var sorted = rows.slice().sort(function (a, b) {
    return a.receivedAt < b.receivedAt ? -1 : a.receivedAt > b.receivedAt ? 1 : 0;
  });
  sorted.forEach(function (r) {
    var cls = classifyOrderRow_(r);
    if (cls.kind === 'skip') {
      summary.skipped[cls.reason]++;
      if (cls.reason === 'test') {
        summary.testRows.push(trim_(r.orderId) + ' — ' + (trim_(r.business) || trim_(r.contact)));
      }
      return;
    }
    var res = applyEvent_(model, {
      type: cls.kind, via: cls.via, orderId: r.orderId, at: r.receivedAt,
      business: r.business, contact: r.contact, phone: r.phone, address: r.address,
      totalItems: r.totalItems, totalPrice: r.totalPrice, pdfUrl: r.pdf,
      imported: true, by: 'ייבוא'
    }, now);
    if (res.skipped) {
      summary.skipped.existing++;
      return;
    }
    summary.activities++;
    if (res.created) summary.created++;
    else if (!touched[res.contact.id]) summary.updated++;
    touched[res.contact.id] = true;
  });
  return { summary: summary, touched: touched };
}

    totalPrice: head.indexOf('סה"כ לתשלום'), shipping: head.indexOf('שיטת אספקה'),
    pdf: head.indexOf('קובץ PDF')
  };
  var out = [];
  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    var o = {};
    Object.keys(idx).forEach(function (k) {
      var v = idx[k] >= 0 ? row[idx[k]] : '';
      o[k] = v instanceof Date ? Utilities.formatDate(v, TZ, 'yyyy-MM-dd HH:mm') : trim_(v);
    });

  }
  return out;
}

function book_() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

// לשונית עם כותרות, נוצרת בפעם הראשונה. כל הכתיבה בפורמט טקסט ('@'):
// אחרת גוגל שיטס מוחק את ה-0 בטלפון והופך תאריכים למספרים.
function sheet_(book, name, headers) {
  var sh = book.getSheetByName(name) || book.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, headers.length).setNumberFormat('@').setValues([headers]);
    sh.setFrozenRows(1);
    sh.setRightToLeft(true);
  }
  return sh;
}

function readRows_(sh, keys) {
  var last = sh.getLastRow();
  if (last < 2) return [];
  return sh.getRange(2, 1, last - 1, keys.length).getValues().map(function (row, i) {
    var o = { _row: i + 2 };
    keys.forEach(function (k, j) { o[k] = unescape_(String(row[j] == null ? '' : row[j])); });
    return o;
  });
}

function readModel_(book) {
  var cs = sheet_(book, CONTACTS_SHEET, CONTACT_HEADERS);
  var as = sheet_(book, ACTIVITIES_SHEET, ACTIVITY_HEADERS);
  return { cs: cs, as: as, contacts: readRows_(cs, CONTACT_KEYS), activities: readRows_(as, ACTIVITY_KEYS) };
}

function rowOf_(o, keys) {
  return keys.map(function (k) {
    return sanitize_(o[k], k === 'quoteLink' ? 5000 : k === 'text' ? 2000 : 300);
  });
}

function writeRows_(sh, startRow, rows, width) {
  if (!rows.length) return;
  sh.getRange(startRow, 1, rows.length, width).setNumberFormat('@').setValues(rows);
}

// אנשי קשר עם _row נכתבים במקומם; חדשים ופעולות חדשות — בסוף הלשונית
function save_(model, changedContacts, newActivities) {
  var fresh = [];
  changedContacts.forEach(function (c) {
    if (c._row) writeRows_(model.cs, c._row, [rowOf_(c, CONTACT_KEYS)], CONTACT_KEYS.length);
    else fresh.push(c);
  });
  var next = model.cs.getLastRow() + 1;
  fresh.forEach(function (c, i) { c._row = next + i; });
  writeRows_(model.cs, next, fresh.map(function (c) { return rowOf_(c, CONTACT_KEYS); }), CONTACT_KEYS.length);
  writeRows_(model.as, model.as.getLastRow() + 1,
    newActivities.map(function (a) { return rowOf_(a, ACTIVITY_KEYS); }), ACTIVITY_KEYS.length);
}

function publicOf_(o, keys) {
  var out = {};
  keys.forEach(function (k) { out[k] = o[k] == null ? '' : String(o[k]); });
  return out;
}

// ---------- פעולות ----------

function handleEvent_(ev) {
  if (['lead', 'quote_request', 'order'].indexOf(ev.type) < 0) return { ok: false, error: 'bad event' };
  var model = readModel_(book_());
  var res = applyEvent_(model, ev, now_());
  if (res.skipped) return { ok: true, id: res.contact ? res.contact.id : '', duplicate: true };
  save_(model, [res.contact], [res.activity]);
  return { ok: true, id: res.contact.id };
}

function handleList_() {
  var model = readModel_(book_());
  return {
    ok: true,
    today: now_().date,
    contacts: model.contacts.map(function (c) { return publicOf_(c, CONTACT_KEYS); }),
    activities: model.activities.map(function (a) { return publicOf_(a, ACTIVITY_KEYS); })
  };
}

function handleAdd_(input) {
  var phone = normalizePhone_(input.phone);
  if (!phone) return { ok: false, error: 'bad phone' };
  var name = trim_(input.name).slice(0, 200);
  var business = trim_(input.business).slice(0, 200);
  if (!name && !business) return { ok: false, error: 'name required' };
  var followUp = trim_(input.followUp);
  if (followUp && !/^\d{4}-\d{2}-\d{2}$/.test(followUp)) return { ok: false, error: 'bad date' };
  var model = readModel_(book_());
  var dup = findContact_(model.contacts, phone, '');
  if (dup) return { ok: false, error: 'duplicate', id: dup.id };
  var now = now_();
  var c = blankContact_(now.stamp);
  c.name = name;
  c.business = business;
  c.phone = phone;
  c.email = trim_(input.email).slice(0, 200);
  c.source = MANUAL_SOURCES.indexOf(trim_(input.source)) >= 0 ? trim_(input.source) : 'אחר';
  c.followUp = followUp;
  model.contacts.push(c);
  var a = {
    id: newId_('A'), contactId: c.id, at: now.stamp, type: 'נוצר ידנית',
    text: trim_(input.note).slice(0, 2000), orderId: '', pdf: '', quoteLink: '', by: 'עמוד'
  };
  save_(model, [c], [a]);
  return { ok: true, contact: publicOf_(c, CONTACT_KEYS), activity: publicOf_(a, ACTIVITY_KEYS) };
}

// בודק את כל השדות לפני שמשנה משהו — שגיאה אחת משאירה את איש הקשר כמו שהיה
function applyFields_(model, c, f) {
  var next = {};
  if ('stage' in f) {
    if (STAGES.indexOf(f.stage) < 0) return { ok: false, error: 'bad stage' };
    next.stage = f.stage;
  }
  if ('followUp' in f) {
    var d = trim_(f.followUp);
    if (d && !/^\d{4}-\d{2}-\d{2}$/.test(d)) return { ok: false, error: 'bad date' };
    next.followUp = d;
  }
  if ('phone' in f) {
    var p = normalizePhone_(f.phone);
    if (!p && trim_(f.phone)) return { ok: false, error: 'bad phone' };
    if (p) {
      var other = findContact_(model.contacts, p, '');
      if (other && other.id !== c.id) return { ok: false, error: 'duplicate', id: other.id };
    }
    next.phone = p;
  }
  ['name', 'business', 'email', 'address'].forEach(function (k) {
    if (k in f) next[k] = trim_(f[k]).slice(0, 200);
  });
  if ('name' in next || 'business' in next) {
    var name = 'name' in next ? next.name : c.name;
    var business = 'business' in next ? next.business : c.business;
    if (!name && !business) return { ok: false, error: 'name required' };
  }
  Object.keys(next).forEach(function (k) { c[k] = next[k]; });
  return { ok: true };
}

function handleUpdate_(id, fields, activity) {
  var model = readModel_(book_());
  var c = contactById_(model, id);
  if (!c) return { ok: false, error: 'not found' };
  if (activity && PAGE_ACTIVITY_TYPES.indexOf(trim_(activity.type)) < 0) {
    return { ok: false, error: 'bad activity' };
  }
  var check = applyFields_(model, c, fields || {});
  if (!check.ok) return check;
  var added = [];
  if (activity) {
    var now = now_();
    added.push({
      id: newId_('A'), contactId: c.id, at: now.stamp, type: trim_(activity.type),
      text: trim_(activity.text).slice(0, 2000), orderId: '', pdf: '', quoteLink: '', by: 'עמוד'
    });
    c.lastActivity = now.stamp;
  }
  save_(model, [c], added);
  return {
    ok: true,
    contact: publicOf_(c, CONTACT_KEYS),
    activity: added.length ? publicOf_(added[0], ACTIVITY_KEYS) : null
  };
}

function handleDelete_(id) {
  var model = readModel_(book_());
  var c = contactById_(model, id);
  if (!c) return { ok: false, error: 'not found' };
  // מלמטה למעלה, כדי שמחיקת שורה לא תזיז את השורות שעוד לא נמחקו
  var rows = model.activities
    .filter(function (a) { return a.contactId === id; })
    .map(function (a) { return a._row; })
    .sort(function (a, b) { return b - a; });
  rows.forEach(function (r) { model.as.deleteRow(r); });
  model.cs.deleteRow(c._row);
  return { ok: true, deleted: rows.length };
}

// מציין מקום (PUT_...) או ערך קצר — חוסם הכל, כדי שהדבקת קובץ הריפו לענן
// לא תפתח את הנתונים
function authorized_(expected, given) {
  if (typeof expected !== 'string' || expected.indexOf('PUT_') === 0 || expected.length < 12) return false;
  return typeof given === 'string' && given === expected;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function withLock_(fn) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return { ok: false, error: 'busy' };
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

// בדיקת חיים בלבד, בלי נתונים
function doGet() {
  return ContentService.createTextOutput('CRM — פעיל');
}

function doPost(e) {
  var raw = (e && e.postData && e.postData.contents) || '';
  if (!raw || raw.length > 100000) return json_({ ok: false, error: 'bad request' });
  var body;
  try {
    body = JSON.parse(raw);
  } catch (err) {
    return json_({ ok: false, error: 'bad request' });
  }
  try {
    if (body.action === 'event') {
      if (!authorized_(CRM_SECRET, body.secret)) return json_({ ok: false, error: 'unauthorized' });
      return json_(withLock_(function () { return handleEvent_(body.event || {}); }));
    }
    if (!authorized_(CRM_PASSWORD, body.password)) return json_({ ok: false, error: 'unauthorized' });
    switch (body.action) {
      case 'list': return json_(handleList_());
      case 'add': return json_(withLock_(function () { return handleAdd_(body.contact || {}); }));
      case 'update': return json_(withLock_(function () { return handleUpdate_(body.id, body.fields || {}, null); }));
      case 'log': return json_(withLock_(function () { return handleUpdate_(body.id, body.fields || {}, body.activity || null); }));
      case 'delete': return json_(withLock_(function () { return handleDelete_(body.id); }));
      case 'import': return json_(withLock_(function () { return handleImport_(body.dryRun !== false); }));
    }
    return json_({ ok: false, error: 'unknown action' });
  } catch (err) {
    Logger.log('CRM: ' + err);
    return json_({ ok: false, error: 'server error' });
  }
}

   node tools/test/crm-page.test.js */
(function (root) {
    'use strict';

    var STAGES = ['חדש', 'חזרתי', 'נשלחה הצעה', 'לקוח', 'לא רלוונטי'];
    var STAGE_SLUGS = { 'חדש': 'new', 'חזרתי': 'back', 'נשלחה הצעה': 'quote', 'לקוח': 'customer', 'לא רלוונטי': 'closed' };
    var FILTERS = [
        { key: 'todo', label: 'לטיפול' },
        { key: 'חדש', label: 'חדש' },
        { key: 'חזרתי', label: 'חזרתי' },
        { key: 'נשלחה הצעה', label: 'נשלחה הצעה' },
        { key: 'לקוח', label: 'לקוח' },
        { key: 'לא רלוונטי', label: 'לא רלוונטי' },
        { key: 'all', label: 'הכל' }
    ];
    var MANUAL_SOURCES = ['וואטסאפ', 'טלפון', 'מודעה בפייסבוק', 'הפניה', 'אחר'];
    var CHANNELS = { call: 'שיחה', whatsapp: 'וואטסאפ', quote: 'הצעת מחיר' };
    var OUTCOMES = [
        { key: 'no_answer', label: 'לא ענה', hint: 'לחזור מחר' },
        { key: 'talked', label: 'דיברנו', hint: 'בוחרים מתי לחזור' },
        { key: 'quote_sent', label: 'שלחתי הצעת מחיר', hint: 'לחזור בעוד 3 ימים' },
        { key: 'ordered', label: 'הזמין', hint: 'הופך ללקוח' },
        { key: 'not_relevant', label: 'לא רלוונטי', hint: 'נסגר' }
    ];
    var FOLLOW_DAYS = { tomorrow: 1, three: 3, week: 7 };

    // ---------- תאריכים (מחרוזות YYYY-MM-DD בזמן מקומי) ----------

    function pad(n) { return (n < 10 ? '0' : '') + n; }
    function ymd(d) { return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()); }
    function parseYmd(s) {
        var p = String(s).split('-');
        return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
    }

    // skipSaturday: תאריך שמחושב מקיצור ויוצא בשבת עובר לראשון
    function addDays(day, n, skipSaturday) {
        var d = parseYmd(day);
        d.setDate(d.getDate() + n);
        if (skipSaturday && d.getDay() === 6) d.setDate(d.getDate() + 1);
        return ymd(d);
    }
    function daysBetween(a, b) { return Math.round((parseYmd(b) - parseYmd(a)) / 86400000); }
    function formatDate(day) {
        var p = String(day || '').split('-');
        return p.length === 3 ? p[2] + '.' + p[1] : '';
    }
    function formatStamp(stamp) {
        var s = String(stamp || '');
        var time = s.slice(11, 16);
        return formatDate(s.slice(0, 10)) + (time ? ' · ' + time : '');
    }
    function relativeDay(day, today) {
        if (!day) return '';
        if (day === today) return 'היום';
        if (day === addDays(today, -1)) return 'אתמול';
        return formatDate(day);
    }
    function nowStamp(d) {
        d = d || new Date();
        return ymd(d) + ' ' + pad(d.getHours()) + ':' + pad(d.getMinutes());
    }

    // ---------- טלפון וקישורים ----------

    // אותם כללים כמו normalizePhone_ בסקריפט ה-CRM
    function normalizePhone(raw) {
        var d = String(raw == null ? '' : raw).replace(/\D/g, '');
        if (d.indexOf('00972') === 0) d = d.slice(2);
        if (d.indexOf('972') === 0) d = d.slice(3);
        if (/^[57]\d{8}$/.test(d) || /^[2-489]\d{7}$/.test(d)) d = '0' + d;
        return /^(0[57]\d{8}|0[2-489]\d{7}|1(?:599|700|800|801)\d{6})$/.test(d) ? d : '';
    }
    function formatPhone(d) {
        d = String(d || '');
        if (/^0[57]\d{8}$/.test(d)) return d.slice(0, 3) + '-' + d.slice(3);
        if (/^0[2-489]\d{7}$/.test(d)) return d.slice(0, 2) + '-' + d.slice(2);
        return d;
    }
    function waLink(phone) {
        var d = String(phone || '');
        return /^05\d{8}$/.test(d) ? 'https://wa.me/972' + d.slice(1) : '';
    }
    // דף הסוכן בוחר את הלקוח לפי customer, ויוצר אותו עם contact ו-phone אם אינו קיים
    function agentLink(c) {
        var q = '?customer=' + encodeURIComponent(c.business || c.name || '');
        if (c.name) q += '&contact=' + encodeURIComponent(c.name);
        if (c.phone) q += '&phone=' + encodeURIComponent(c.phone);
        return '/agent/' + q;
    }
    function displayName(c) { return c.business || c.name || formatPhone(c.phone) || 'ללא שם'; }
    function stageSlug(stage) { return STAGE_SLUGS[stage] || 'new'; }

    // ---------- רשימה ----------

    // תאריך חזרה שהגיע, או "חדש" בלי תאריך — חוץ מ"לא רלוונטי". ליד חדש עם
    // תאריך עתידי מחכה לתאריך שלו.
    function isTodo(c, today) {
        if (c.stage === 'לא רלוונטי') return false;
        if (c.followUp) return c.followUp <= today;
        return c.stage === 'חדש';
    }
    function isOverdue(c, today) { return !!c.followUp && c.followUp < today && c.stage !== 'לא רלוונטי'; }
    function todoGroup(c, today) {
        if (c.stage === 'חדש') return 0;
        return isOverdue(c, today) ? 1 : 2;
    }
    function matchesQuery(c, query) {
        var q = String(query || '').trim().toLowerCase();
        if (!q) return true;
        if ([c.name, c.business, c.email].join(' ').toLowerCase().indexOf(q) >= 0) return true;
        var digits = q.replace(/\D/g, '');
        if (digits.length < 3) return false;
        if (digits.indexOf('972') === 0) digits = '0' + digits.slice(3);
        var phone = String(c.phone || '');
        return phone.indexOf(digits) >= 0 || phone.replace(/^0/, '').indexOf(digits) >= 0;
    }
    function cmp(a, b) { return a < b ? -1 : a > b ? 1 : 0; }
    // "לטיפול": חדשים (החדש למעלה), באיחור (הישן למעלה), של היום. אחרים: פעולה אחרונה
    function sortContacts(list, filter, today) {
        return list.slice().sort(function (a, b) {
            if (filter === 'todo') {
                var ga = todoGroup(a, today), gb = todoGroup(b, today);
                if (ga !== gb) return ga - gb;
                if (ga === 0) return cmp(b.created, a.created);
                if (ga === 1) return cmp(a.followUp, b.followUp);
            }
            return cmp(b.lastActivity || b.created, a.lastActivity || a.created);
        });
    }
    function filterContacts(contacts, filter, query, today) {
        var list = contacts.filter(function (c) {
            if (!matchesQuery(c, query)) return false;
            if (filter === 'todo') return isTodo(c, today);
            if (filter === 'all') return true;
            return c.stage === filter;
        });
        return sortContacts(list, filter, today);
    }
    function counts(contacts, today) {
        var r = { todo: 0, all: contacts.length };
        STAGES.forEach(function (s) { r[s] = 0; });
        contacts.forEach(function (c) {
            if (r[c.stage] != null) r[c.stage]++;
            if (isTodo(c, today)) r.todo++;
        });
        return r;
    }

    // ---------- תאריך חזרה ----------

    function followUpFor(choice, today) {
        if (choice === 'today') return today;
        if (!FOLLOW_DAYS[choice]) return '';
        return addDays(today, FOLLOW_DAYS[choice], true);
    }
    function followUpValue(c, today) {
        if (!c.followUp) return '';
        if (c.followUp < today) {
            var n = daysBetween(c.followUp, today);
            return n === 1 ? 'באיחור יום' : 'באיחור ' + n + ' ימים';
        }
        if (c.followUp === today) return 'היום';
        if (c.followUp === addDays(today, 1)) return 'מחר';
        return formatDate(c.followUp);
    }
    function followUpLabel(c, today) {
        var v = followUpValue(c, today);
        if (!v) return '';
        return v.indexOf('באיחור') === 0 ? 'לחזור — ' + v : 'לחזור ' + v;
    }
    function subLine(c, today) {
        var parts = [];
        if (c.business && c.name) parts.push(c.name);
        if (c.source) parts.push(c.source);
        var when = relativeDay(String(c.lastActivity || c.created || '').slice(0, 10), today);
        if (when) parts.push(when);
        return parts.join(' · ');
    }

    function outcomesFor(channel) {
        if (channel !== 'quote') return OUTCOMES.slice();
        var first = OUTCOMES.filter(function (o) { return o.key === 'quote_sent'; });
        return first.concat(OUTCOMES.filter(function (o) { return o.key !== 'quote_sent'; }));
    }
    function applyOutcome(c, key, channel, today, follow, note) {
        var fields = {};
        var label = '';
        OUTCOMES.forEach(function (o) { if (o.key === key) label = o.label; });
        if (key === 'no_answer') {
            if (c.stage === 'חדש') fields.stage = 'חזרתי';
            fields.followUp = addDays(today, 1, true);
        } else if (key === 'talked') {
            if (c.stage === 'חדש') fields.stage = 'חזרתי';
            fields.followUp = followUpFor(follow || 'tomorrow', today);
        } else if (key === 'quote_sent') {
            if (c.stage !== 'לקוח') fields.stage = 'נשלחה הצעה';
            fields.followUp = addDays(today, 3, true);
        } else if (key === 'ordered') {
            fields.stage = 'לקוח';
            fields.followUp = '';
        } else if (key === 'not_relevant') {
            fields.stage = 'לא רלוונטי';
            fields.followUp = '';
        }
        return {
            fields: fields,
            activity: { type: CHANNELS[channel] || 'שיחה', text: label + (note ? ' — ' + note : '') }
        };
    }

    // ---------- טופס ----------

    function findByPhone(contacts, raw) {
        var p = normalizePhone(raw);
        if (!p) return null;
        for (var i = 0; i < contacts.length; i++) if (contacts[i].phone === p) return contacts[i];
        return null;
    }
    function validateNew(input, mode) {
        var raw = String(input.phone || '').trim();
        if (mode !== 'edit' && !raw) return 'צריך מספר טלפון.';
        if (raw && !normalizePhone(raw)) return 'מספר הטלפון לא תקין — למשל 050-1234567.';
        if (!String(input.name || '').trim() && !String(input.business || '').trim()) return 'צריך שם או שם עסק.';
        return '';
    }

    // ---------- ייבוא ----------

    function importLines(s, done) {
        var lines = [];
        if (!done && !s.activities) {
            lines.push('אין מה לייבא — הכל כבר כאן.');
        } else {
            lines.push((done ? 'יובאו ' : 'ייובאו ') + s.activities + ' הזמנות ובקשות: ' +
                s.created + ' אנשי קשר חדשים, ' + s.updated + ' קיימים מתעדכנים.');
        }
        var sk = s.skipped || {};
        var names = [['chain', 'רשת החנויות'], ['committee', 'ועד העובדים'], ['consumer', 'הזמנות צרכן'],
            ['test', 'שורות בדיקה'], ['noKey', 'בלי טלפון ובלי עסק'], ['unknown', 'קידומת לא מוכרת'],
            ['existing', 'כבר קיימים']];
        var parts = names.filter(function (n) { return sk[n[0]]; })
            .map(function (n) { return n[1] + ' ' + sk[n[0]]; });
        if (parts.length) lines.push('דולגו: ' + parts.join(' · ') + '.');
        return lines;
    }

    var CrmCore = {
        STAGES: STAGES, FILTERS: FILTERS, MANUAL_SOURCES: MANUAL_SOURCES, CHANNELS: CHANNELS,
        OUTCOMES: OUTCOMES, ymd: ymd, addDays: addDays, daysBetween: daysBetween,
        formatDate: formatDate, formatStamp: formatStamp, relativeDay: relativeDay,
        nowStamp: nowStamp, normalizePhone: normalizePhone, formatPhone: formatPhone,
        waLink: waLink, agentLink: agentLink, displayName: displayName, stageSlug: stageSlug,
        isTodo: isTodo, isOverdue: isOverdue, filterContacts: filterContacts, counts: counts,
        followUpFor: followUpFor, followUpValue: followUpValue, followUpLabel: followUpLabel,
        subLine: subLine, outcomesFor: outcomesFor, applyOutcome: applyOutcome,
        findByPhone: findByPhone, validateNew: validateNew, importLines: importLines
    };
    if (typeof module !== 'undefined' && module.exports) module.exports = CrmCore;
    else root.CrmCore = CrmCore;
})(this);
